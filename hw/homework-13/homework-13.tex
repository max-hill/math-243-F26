\begin{problem}
  Let $f(x,y)=\sin(x^2-y)$. Find the unit vector $\boldsymbol{u}$ that maximizes
  $D_{\boldsymbol{u}}f(1,1)$, and find $D_{\boldsymbol{u}}f(1,1)$.
\end{problem}
\vfill
\begin{problem}
  A heat-seeking missile has the property that any point $(x,y)$ in the plane,
  it moves in the direction of maximum temperature increase. Assume the
  temperature at each point $(x,y)$ is given by
  \begin{equation*}
    T(x,y) = -e^{-2y}\cos(x).
  \end{equation*}
  When the missile is at the point $(x,y) = (\pi/4,0)$, what direction is it pointing?
\end{problem}
\vfill
\newpage

\begin{problem}
  Find the local maxima, local minima, and saddle points of the functions:
  \begin{enumerate}[(a)]
    \item $f(x,y) = x^{2}+xy+y^{2}+3x-3y+4 $
    \item $f(x,y) = 2xy - 5x^{2} - 2y^{2} + 4x +4y -4$
  \end{enumerate}
\end{problem}
\newpage


\begin{problem}
  Find the absolute maxima and minima of the function $f(x,y) = 2x^{2}-4x+y^{2}-4y+1$
  on the closed triangular region bounded by the lines
  \begin{equation*}
    x=0, \qquad y = 2, \qquad y=2x.
  \end{equation*}
  Begin by sketching the triangular region.
\end{problem}
\vfill
\begin{problem}
  Find the global max and min of the function $f(x,y) =x^{2}-2xy+2y$ on
  \begin{equation*}
    D = \left\{(x,y)\in \R^{2}: 0 \leq x \leq 3, 0 \leq y \leq 2\right\}
  \end{equation*}
\end{problem}
\vfill
\newpage

\begin{problem}
  Let $f(x,y) = x^{2}+y^{2}$. Find the global maximum and minimum of $f$ on
  the closed region between the line $y=-2$ and the parabola $ y=2-x^{2}$.
\end{problem}

\vfill
\begin{problem}
  Find the absolute max and min values of $f(x,y) = x^{2}+xy+y^{2}-3x+3y$ on
  the triangular region cut from the first quadrant by the line $x+y=4$.
\end{problem}
