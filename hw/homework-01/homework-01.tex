\begin{problem}[Arc length]

  $x=2$. Show your work. 
\end{problem}



\begin{problem}
  Consider the following parametric equations
  \begin{equation*}
    \left\{ \begin{array}{l@{}l} x &=3\cos(t)+4 \\ y&= \sin(t) +2\end{array}\right.
  \end{equation*}
  for all $t$ in the interval $[0,2\pi]$.
  \begin{enumerate}[(a.)]
    \item Fill in the following table values, and plot the points on a graph.
    \begin{center}
      \resizebox{.4\textwidth}{!}{%
        \begin{tabular}{lll}
          \hline
          \multicolumn{1}{|l|}{$t$} & \multicolumn{1}{l|}{$x(t)$} & \multicolumn{1}{l|}{$y(t)$} \\ \hline
          \multicolumn{1}{|l|}{$0$} & \multicolumn{1}{l|}{} & \multicolumn{1}{l|}{} \\ \hline
          \multicolumn{1}{|l|}{$\pi/6$}  & \multicolumn{1}{l|}{} & \multicolumn{1}{l|}{} \\ \hline
          \multicolumn{1}{|l|}{ $\pi/4 $ }  & \multicolumn{1}{l|}{} & \multicolumn{1}{l|}{} \\ \hline
          \multicolumn{1}{|l|}{ $\pi/3 $ }  & \multicolumn{1}{l|}{} & \multicolumn{1}{l|}{} \\ \hline
          \multicolumn{1}{|l|}{ $\pi/2 $ }  & \multicolumn{1}{l|}{} & \multicolumn{1}{l|}{} \\ \hline
          \multicolumn{1}{|l|}{ $2\pi/3 $ } & \multicolumn{1}{l|}{} & \multicolumn{1}{l|}{} \\ \hline
          \multicolumn{1}{|l|}{ $3\pi/4 $ } & \multicolumn{1}{l|}{} & \multicolumn{1}{l|}{} \\ \hline
          \multicolumn{1}{|l|}{ $5\pi/6 $ } & \multicolumn{1}{l|}{} & \multicolumn{1}{l|}{} \\ \hline
          \multicolumn{1}{|l|}{$\pi$} & \multicolumn{1}{l|}{} & \multicolumn{1}{l|}{} \\ \hline
        \end{tabular}
      }
    \end{center}
    
    \item Convert the parametric equations to Cartesian form by eliminating the variable $t$.
    \textit{(Hint: use the trig identity $\sin^{2}(t)+\cos^{2}(t) =1$.)}
    \item Sketch the graph of the curve as $t$ ranges from $0$ to $2\pi$.
  \end{enumerate}
\end{problem}
