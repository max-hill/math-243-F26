\begin{problem}
  Consider the parametric equations
  \begin{equation*}
    \left\{ \begin{array}{l@{}l} x=2t-5 \\ y=4t-7 \end{array}\right., \qquad -\infty<t<\infty.
  \end{equation*}
  \begin{enumerate}[(a)]
    \item Identify the ant's path by finding a Cartesian equation for it.
    \vspace{2in}
    \item The \demph{vector equation of a line} is
    \begin{equation*}
      \mathbf{r}(t) = \mathbf{v}t+\mathbf{b}, \qquad -\infty<t<\infty
    \end{equation*}
    where
    \begin{equation*}
      \mathbf{v}=
      \begin{bmatrix}
        v_{1}\\v_{2}
      \end{bmatrix}
      \qquad  \mathbf{b}=
      \begin{bmatrix}
        b_{1}\\b_{2}
      \end{bmatrix}.    
    \end{equation*}
    \textit{(Note the similarity to the familiar line equation $y=mx+b$.)}

    Write the ant's path in this form, and use it to identify the direction of
    the ant's travel, and its position at $t=0$.
    \vspace{1in}
  \end{enumerate}
\end{problem}

\begin{problem}
  An ant walks around the ellipse
  \[
    \mathbf r(t)=(4\cos t+3,\,2\sin t+1),
    \qquad 0\leq t\leq2\pi.
  \]
  Compare the ant's speed at $t=0$ with its speed at $t=\pi/2$. Does the ant
  travel around the ellipse at constant speed?
\end{problem}
\vspace{1.5in}

\begin{problem}
  \begin{enumerate}[(a)]
    \item[]
    \item Give an example of a parametric curve that cannot be written as the
    form in Cartesian form $y=f(x)$.
    \vspace{1in}
    \item \textbf{(True/False)} Every curve that can be written in the form
    $y=f(x)$ can also be described by parametric equations. Justify your
    answer.
    \vspace{1in}
    \item Theorem 1 from the lecture notes says that the graph of $y=f(x)$ over
    $[a,b]$ has arc length
    \begin{equation*}
      (\text{arc length})=\int_{a}^{b}\sqrt{1+\left[ f'(x) \right]^{2} }dx.
    \end{equation*}
    Show that this formula is implied by Theorem 18 from the lecture notes.
    (Hint: consider the parameterization $\mathbf{r}(t) = (t, f(t))$,
    $a \leq t \leq b$. \textit{[This shows that Theorem~1 is just a special
      case of Theorem~18].}
    \vspace{1in}
  \end{enumerate}
\end{problem}

\begin{problem}
  In Example 15 of the lectures, we eliminated the parameter from
  \begin{equation*}
    \left\{ \begin{array}{l@{}l} x=5\cos^{2}(\theta) \\  y=2\sin^{2}(\theta)\end{array}\right.
  \end{equation*}
  and obtained the line $y=-\frac{2}{5}x +2$. Explain in a sentence or two \emph{what
    information was lost} in that computation, and why the restriction
  $0\le x\le 5$ had to be added by hand.
  \vspace{1in}
\end{problem}

\begin{problem}
  Explain why the ellipse equation
  $\left(\frac{x-h}{a}\right)^{2}+\left(\frac{y-k}{b}\right)^{2}=1$ cannot, by
  itself, tell you the direction or speed of a particle traveling around the
  ellipse, whereas the parametric form can.
  \vspace{1in}
\end{problem}

\begin{problem}
  Suppose we have a parameterized curve $\mathbf{r}(t)=(x(t),y(t))$ such
  that at some time $t_{0}$ we have $x'(t_{0})=0$ but $y'(t_{0})\neq0$. What
  does this tell you about the ant's motion at that instant, and what does it
  tell you about the shape of the curve there?
  \vspace{1in}
\end{problem}

\begin{problem}
  The distance traveled by an ant along a curve $\mathbf{r}(t),$ $a \leq
  t \leq b$ is
  \begin{equation*}
    (\text{distance traveled}) = \int_{a}^{b}\left|\mathbf r'(t)\right|dt
  \end{equation*}
  Why do we integrate $|\mathbf{r}'(t)|$ rather than $\mathbf{r}(t)$? What
  would $\int_{a}^{b}\mathbf r'(t)\,dt$ compute instead, and why is that a
  different quantity?
  \vspace{1in}
\end{problem}



\begin{problem}
  During the short time interval $[t,t+\Delta t]$, we approximate the ant's distance
  traveled by 
  \[
    (\text{distance traveled})\approx
    |\mathbf r'(t)|\Delta t
  \]
  This formula pretends that the ant travels at the constant speed
  $|\mathbf r'(t)|$ throughout the interval, it's generally not the exact
  distance traveled. Explain why this approximation improves as the interval
  gets shorter. \vspace{1.2in}
\end{problem}
